rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================
    // 共通
    // =========================

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    function isPersonalGroup(groupId) {
      return signedIn() && groupId == ("PERSONAL_" + request.auth.uid);
    }

    function groupExists(groupId) {
      return exists(/databases/$(database)/documents/groups/$(groupId));
    }

    function groupData(groupId) {
      return get(/databases/$(database)/documents/groups/$(groupId)).data;
    }

    function isGroupMember(groupId) {
      return signedIn()
        && groupExists(groupId)
        && request.auth.uid in groupData(groupId).members;
    }

    function isGroupCreator(groupId) {
      return signedIn()
        && groupExists(groupId)
        && groupData(groupId).createdBy == request.auth.uid;
    }

    function isGroupBanned(groupId) {
      return signedIn()
        && groupExists(groupId)
        && request.auth.uid in groupData(groupId).bannedUids;
    }

    function canAccessGroup(groupId) {
      return isAdmin()
        || isPersonalGroup(groupId)
        || (isGroupMember(groupId) && !isGroupBanned(groupId));
    }


    // =========================
    // users
    // =========================

    match /users/{userId} {

      // グループメンバーの表示名を取得するため、
      // ログイン済みユーザーは他ユーザーのプロフィールも取得可能。
      // ※ users ドキュメントにはメールアドレス等も含まれるため、
      // 将来的には publicProfiles/{userId} へ表示名を分離するのがより安全。
      allow read: if signedIn();

      // 自分のアカウント作成のみ
      // role を勝手に admin に変更できないようにする
      allow create: if signedIn()
                    && request.auth.uid == userId
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.role == "user";

      // 自分のアカウント更新のみ
      // role は管理者だけ変更可能
      allow update: if signedIn()
                    && request.auth.uid == userId
                    && request.resource.data.uid == resource.data.uid
                    && request.resource.data.role == resource.data.role;

      // ユーザー削除は管理者のみ
      allow delete: if isAdmin();
    }


    // =========================
    // groups
    // =========================

    match /groups/{groupId} {

      // BAN判定をここに入れると、
      // members/createdBy を条件にしたFirestore queryが
      // 「そのqueryが常に許可される」と判定できず
      // Missing or insufficient permissions になる場合があるため、
      // groupドキュメントのreadは membership/creator/admin で判定する。
      //
      // BAN済みユーザーは members から除外されるため、
      // 通常はこのread条件にも一致しない。
      allow read: if signedIn()
                  && (
                    isAdmin()
                    || resource.data.createdBy == request.auth.uid
                    || request.auth.uid in resource.data.members
                  );

      // 作成時は作成者が初期メンバー
      allow create: if signedIn()
                    && request.resource.data.createdBy == request.auth.uid
                    && request.resource.data.members is list
                    && request.auth.uid in request.resource.data.members
                    && request.resource.data.bannedUids is list
                    && !(request.auth.uid in request.resource.data.bannedUids)
                    && request.resource.data.code is string;

      // 管理者・作成者: 全更新可
      allow update: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.createdBy == request.auth.uid
                    )
                    // 一般メンバーの参加:
                    // 既存データを変更せず、自分自身だけを1人追加できる。
                    // ただしBAN済みユーザーはルール上も再参加不可。
                    || (
                      signedIn()
                      && !(request.auth.uid in resource.data.bannedUids)
                      && request.resource.data.createdBy == resource.data.createdBy
                      && request.resource.data.code == resource.data.code
                      && request.resource.data.name == resource.data.name
                      && request.resource.data.bannedUids == resource.data.bannedUids
                      && request.resource.data.members is list
                      && resource.data.members is list
                      && request.resource.data.members.hasAll(resource.data.members)
                      && request.resource.data.members.size() == resource.data.members.size() + 1
                      && request.auth.uid in request.resource.data.members
                      && !(request.auth.uid in resource.data.members)
                    )
                    // 一般メンバーの退会:
                    // 自分自身だけを1人削除できる。
                    || (
                      signedIn()
                      && request.resource.data.createdBy == resource.data.createdBy
                      && request.resource.data.code == resource.data.code
                      && request.resource.data.name == resource.data.name
                      && request.resource.data.bannedUids == resource.data.bannedUids
                      && request.resource.data.members is list
                      && resource.data.members is list
                      && resource.data.members.hasAll(request.resource.data.members)
                      && request.resource.data.members.size() == resource.data.members.size() - 1
                      && request.auth.uid in resource.data.members
                      && !(request.auth.uid in request.resource.data.members)
                    );

      // 作成者または管理者のみ削除
      allow delete: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.createdBy == request.auth.uid
                    );
    }


    // =========================
    // groupInvites
    // =========================

    match /groupInvites/{inviteId} {
      // inviteId を知っている場合のみ取得を許可（一覧は禁止）
      allow get: if signedIn();
      allow list: if false;

      // 作成者/管理者のみ管理
      allow create, update, delete: if isAdmin()
                                    || (
                                      signedIn()
                                      && request.resource.data.createdBy == request.auth.uid
                                    );
    }


    // =========================
    // schedules
    // =========================

    match /schedules/{scheduleId} {

      // グループアクセス権がある予定のみ閲覧
      allow read: if signedIn()
                  && resource.data.groupId is string
                  && canAccessGroup(resource.data.groupId);

      // 自分の予定として作成
      // グループのメンバーであることが必須
      allow create: if signedIn()
                    && request.resource.data.userId == request.auth.uid
                    && request.resource.data.groupId is string
                    && request.resource.data.date is string
                    && request.resource.data.time is string
                    && request.resource.data.status in ["ok", "maybe", "ng"]
                    && canAccessGroup(request.resource.data.groupId);

      // 自分の予定だけ変更可能
      // 管理者は全予定変更可能
      allow update: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.userId == request.auth.uid
                      && request.resource.data.userId == resource.data.userId
                      && request.resource.data.groupId == resource.data.groupId
                      && canAccessGroup(resource.data.groupId)
                    );

      // 自分の予定または管理者のみ削除
      allow delete: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.userId == request.auth.uid
                      && canAccessGroup(resource.data.groupId)
                    );
    }


    // =========================
    // reviews
    // =========================

    match /reviews/{reviewId} {

      // レビューは誰でも閲覧可能
      allow read: if true;

      // ログインユーザーが自分のレビューを投稿
      allow create: if signedIn()
                    && request.resource.data.userId == request.auth.uid
                    && request.resource.data.rating is int
                    && request.resource.data.rating >= 1
                    && request.resource.data.rating <= 5
                    && request.resource.data.comment is string
                    && request.resource.data.author is string;

      // 自分のレビューまたは管理者のみ変更
      allow update: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.userId == request.auth.uid
                      && request.resource.data.userId == resource.data.userId
                    );

      // 自分のレビューまたは管理者のみ削除
      allow delete: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.userId == request.auth.uid
                    );
    }


    // =========================
    // settings
    // =========================

    match /settings/{settingId} {

      // お知らせは全員閲覧可能
      allow read: if true;

      // 設定変更は管理者だけ
      allow write: if isAdmin();
    }


    // =========================
    // その他
    // =========================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
